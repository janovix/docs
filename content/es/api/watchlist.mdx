# API de Watchlist

**URL base:** `https://watchlist.janovix.com`

La API de Watchlist es un servicio independiente de la API AML. Todas las solicitudes deben incluir tu llave de API (la misma llave funciona para ambos servicios).

## Endpoints

| Método | Ruta                | Descripción                                                 |
| ------ | ------------------- | ----------------------------------------------------------- |
| `POST` | `/search`           | Búsqueda híbrida en todos los conjuntos de datos            |
| `POST` | `/search/ofac`      | Búsqueda solo en OFAC                                       |
| `POST` | `/search/unsc`      | Búsqueda solo en UNSC                                       |
| `POST` | `/search/sat69b`    | Búsqueda solo en SAT 69-B                                   |
| `GET`  | `/queries`          | Listar consultas anteriores                                 |
| `GET`  | `/queries/:queryId` | Obtener consulta con resultados completos                   |
| `GET`  | `/events`           | Flujo SSE para resultados asíncronos (PEP, Medios Adversos) |

```typescript
const WATCHLIST_URL = "https://watchlist.janovix.com";

interface SearchInput {
query: string;
datasets?: ("ofac_sdn" | "unsc" | "sat_69b" | "pep" | "adverse_media")[];
entityType?: "person" | "organization";
topK?: number; // predeterminado: 5
threshold?: number; // 0–1, predeterminado: 0.6
birthDate?: string; // YYYY-MM-DD para mayor precisión
identifiers?: { type: string; value: string }[];
}

async function screenName(input: SearchInput) {
const res = await fetch(`${WATCHLIST_URL}/search`, {
method: "POST",
headers: {
Authorization: `Bearer ${process.env.JANOVIX_API_KEY}`,
"Content-Type": "application/json",
},
body: JSON.stringify(input),
});
return res.json();
}

const results = await screenName({
query: "Juan García Martínez",
datasets: ["ofac_sdn", "unsc", "sat_69b"],
entityType: "person",
threshold: 0.65,
});
````

```php
function watchlistSearch(string $query, array $options = []): array {
    global $apiKey;
    $payload = array_merge(["query" => $query], $options);
    $ch = curl_init("https://watchlist.janovix.com/search");
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => json_encode($payload),
        CURLOPT_HTTPHEADER     => [
            "Authorization: Bearer {$apiKey}",
            "Content-Type: application/json",
        ],
    ]);
    $response = curl_exec($ch);
    curl_close($ch);
    return json_decode($response, true);
}

$results = watchlistSearch("Juan García Martínez", [
    "datasets"  => ["ofac_sdn", "unsc", "sat_69b"],
    "threshold" => 0.65,
]);
````

<Callout type="info">
	El `queryId` se devuelve de inmediato. Los resultados de PEP y medios adversos

	flujo SSE en `GET /events` para recibir los resultados cuando estén listos.

```typescript
async function pollQuery(queryId: string, maxAttempts = 10) {
  for (let i = 0; i < maxAttempts; i++) {
    const res = await fetch(`${WATCHLIST_URL}/queries/${queryId}`, {
      headers: { Authorization: `Bearer ${process.env.JANOVIX_API_KEY}` },
    });
    const data = await res.json();
    if (data.status === "complete") return data;
    await new Promise((r) => setTimeout(r, 3000)); // esperar 3s
  }
  throw new Error("La consulta agotó el tiempo de espera");
}
```
